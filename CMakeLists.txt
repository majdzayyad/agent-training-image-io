cmake_minimum_required(VERSION 3.20)
project(adaptive_image_io VERSION 0.1.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(AIO_SANITIZE "Enable address and undefined-behavior sanitizers" OFF)
option(AIO_TSAN "Enable the thread sanitizer (separate build)" OFF)
if(AIO_SANITIZE AND AIO_TSAN)
  message(FATAL_ERROR "Select only one sanitizer configuration")
endif()

find_package(Threads REQUIRED)
find_package(CURL REQUIRED)
add_library(adaptive_io src/remote_backend.cpp src/image_reader.cpp src/fetch_policy.cpp)
target_include_directories(adaptive_io PUBLIC include)
target_link_libraries(adaptive_io PUBLIC Threads::Threads CURL::libcurl)
target_compile_options(adaptive_io PRIVATE -Wall -Wextra -Wpedantic -Werror)
if(AIO_SANITIZE)
  target_compile_options(adaptive_io PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(adaptive_io PUBLIC -fsanitize=address,undefined)
elseif(AIO_TSAN)
  target_compile_options(adaptive_io PUBLIC -fsanitize=thread -fno-omit-frame-pointer)
  target_link_options(adaptive_io PUBLIC -fsanitize=thread)
endif()

add_executable(replay benchmarks/replay.cpp)
target_link_libraries(replay PRIVATE adaptive_io)
target_compile_options(replay PRIVATE -Wall -Wextra -Wpedantic -Werror)

include(CTest)
if(BUILD_TESTING)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  if(AIO_TSAN)
    find_program(SETARCH_EXECUTABLE setarch REQUIRED)
    set(AIO_TEST_PREFIX ${SETARCH_EXECUTABLE} ${CMAKE_SYSTEM_PROCESSOR} -R)
  endif()
  add_executable(io_tests tests/io_tests.cpp)
  target_link_libraries(io_tests PRIVATE adaptive_io)
  target_compile_options(io_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
  add_test(NAME io_correctness COMMAND ${AIO_TEST_PREFIX} $<TARGET_FILE:io_tests>)
  set_tests_properties(io_correctness PROPERTIES TIMEOUT 60)
  add_test(NAME http_contract COMMAND ${AIO_TEST_PREFIX} ${Python3_EXECUTABLE}
    ${CMAKE_CURRENT_SOURCE_DIR}/tests/http_tests.py $<TARGET_FILE:replay>)
  set_tests_properties(http_contract PROPERTIES TIMEOUT 60)
  add_test(NAME experiment_contract COMMAND ${Python3_EXECUTABLE}
    ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_experiment.py)
  set_tests_properties(experiment_contract PROPERTIES TIMEOUT 60)
endif()
